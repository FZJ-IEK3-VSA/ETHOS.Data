% Level 2 of the maintainer side. The boundary is the `ethos-data catalog` group
% and nothing else: `dispatch()` in maintain/cli.py routes build, publish, upload
% and check-store, and has no branch that reaches the namespace builder. That
% builder is drawn OUTSIDE the boundary, with the command that does reach it,
% because a figure that keeps it inside is the one thing a reader would take from
% this page that the code no longer supports -- and it contradicts the paragraph
% printed beside it.
\begin{tikzpicture}
\node[actor, text width=37mm, minimum height=15mm] (manifest) at (0,0) {Manifest builder};
\node[actor, text width=37mm, minimum height=15mm] (upload) at (6,0) {Uploader};
\node[actor, text width=37mm, minimum height=15mm] (publish) at (0,-3) {Public-view generator};
\node[artifact, text width=97mm, minimum height=13mm] (helpers) at (3,-5.5) {Maintainer helpers\\catalogue roots / dataset layout / resource expansion};
\node[artifact, text width=37mm, minimum height=15mm] (matching) at (-6,0) {Selection / Catalogue model\\shared rules};
\node[artifact, text width=37mm, minimum height=15mm] (public) at (-6,-3) {Public catalogue\\filesystem output};
\node[artifact, text width=37mm, minimum height=15mm] (remote) at (12,0) {rclone / oidc-agent\\dCache HTTP interfaces\\external};
% Beside the boundary rather than inside it, and at the height of the commands it
% is a peer of: it reads the same checkout and the same helpers, but it is
% reached from the top-level `link`, never from `catalog`.
\node[actor, text width=37mm, minimum height=15mm] (namespace) at (12,-3) {Namespace builder};

\node[cmd, above=2mm of namespace] (namespace-c) {ethos-data link -{}-all};

\end{tikzpicture}
