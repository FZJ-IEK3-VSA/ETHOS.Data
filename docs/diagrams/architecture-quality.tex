% Quality tree: the six quality goals of section 1, in priority order, and
% the scenarios of section 10 that make each one checkable. One leaf per
% scenario; a leaf belongs to exactly one goal.
%
% Leaves are placed on a 7 mm pitch with a wider gap between goals; each goal
% sits level with the middle of its leaves, and the root level with the
% middle of the goals. Coordinates are millimetres; the picture is 162 wide.
\begin{tikzpicture}[
  x=1mm, y=1mm,
  root/.style={c4system, text width=14mm, minimum height=14mm,
    font=\sffamily\small\bfseries},
  goal/.style={c4container, text width=30mm, minimum height=9mm,
    inner sep=1.4mm, font=\sffamily\footnotesize\bfseries},
  leaf/.style={c4component, text width=95.8mm, minimum height=5.6mm,
    inner sep=1.1mm, align=left, font=\sffamily\footnotesize,
    anchor=west},
  tie/.style={link},
]
\newcommand{\q}[1]{\makebox[8mm][l]{\bfseries #1}}

% ---- leaves --------------------------------------------------------------
% 1 Traceable, repeatable inputs
\node[leaf] (q2)  at (64,0)      {\q{Q2}A corrupt cached file is refused and downloaded again};

\node[leaf] (q10) at (64,-14)    {\q{Q10}A rerun on an older public release reads the same keys and objects};
\node[leaf] (q17) at (64,-21)    {\q{Q17}\texttt{fetch=False} contacts no store and names each missing file};
\node[leaf] (q21) at (64,-28)    {\q{Q21}A handle keeps the catalogue and roots it was built with};
% 2 Controlled access and publication
\node[leaf] (q3)  at (64,-38.5)  {\q{Q3}Restricted data without an installation fails before any download};

% 5 Scalable metadata access

\node[leaf] (q14) at (64,-136.5) {\q{Q14}A new key changes validation, schema and reference together};
\node[leaf] (q15) at (64,-143.5) {\q{Q15}An import from a higher layer fails a test};
\node[leaf] (q16) at (64,-150.5) {\q{Q16}Readers and maintainers refuse a missing shard with one error};
