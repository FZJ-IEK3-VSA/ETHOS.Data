% Shared TikZ styles for every ethos-data diagram.
%
% Rendered twice, once per theme: docs/diagrams/render.py sets \icetwodark before
% \input-ing this file, and the palette below switches. That is necessary rather
% than tidy -- pdftocairo converts text to vector paths, so there is no text left
% in the SVG for CSS to recolour, and a single render would be unreadable in one
% of the two themes. The pages pick between the pair with Material's
% "#only-light" / "#only-dark" URL-fragment convention.
%
% Palette is the FZJ one already used by docs/stylesheets/extra.css:
%   #023d6b  primary blue      #35b0ab  accent teal
% plus greys chosen to keep contrast on both grounds.

\usepackage{tikz}
\usetikzlibrary{positioning,shapes.geometric,arrows.meta,calc,fit,backgrounds}
\usepackage{xcolor}
\usepackage[T1]{fontenc}

% Diagram labels must never hyphenate: "Check and re-pair the cache" is how a
% short label inside a fixed-width node breaks, and it reads as a different word.

% -- C4 vocabulary for the architecture views ---------------------------
% Elements follow the C4 model (c4model.com): people, software systems,
% containers, components and data stores, each with a name, a [kind:
% technology] line and a short responsibility. Write their text with \element:
%   \node[c4container, text width=44mm] (lib)
%     {\element{ethos\_data library}{Container: Python package}{Selects, locates and fetches files}};
% Blue is ETHOS.Data, teal its components, grey everything outside it.
\newcommand{\element}[3]{%
  {\bfseries #1}\\[-0.3ex]{\scriptsize\color{muted}[#2]}\\[0.2ex]{\scriptsize #3}}
% An element with a name and kind only, for crowded views.
\newcommand{\elementshort}[2]{{\bfseries #1}\\[-0.3ex]{\scriptsize\color{muted}[#2]}}
% A relationship label: what is exchanged, then the technology, if any.
\newcommand{\rel}[2]{#1\ifx\relax#2\relax\else\\{\color{muted}[#2]}\fi}

% Key entries, placed inside a scope shifted to where the key goes. Row 0
% holds the title; entries go in rows 1, 2, ...
%   \begin{scope}[shift={(150mm,-10mm)}]
%     \keytitle{Key}

%     \keystore{3}{brand}{fillbrand}{Data store}

\newcommand{\keytitle}[1]{%
  \node[anchor=west, text=muted, font=\sffamily\scriptsize\bfseries, inner sep=0pt]
    at (0,0) {#1};}
\newcommand{\keynode}[3]{%
  \node[#2, keysample] at (4.5mm,{-#1*6.5mm}) {};%
  \node[keytext] at (11.5mm,{-#1*6.5mm}) {#3};}
\newcommand{\keyline}[3]{%
  \draw[#2] (0,{-#1*6.5mm}) -- (9mm,{-#1*6.5mm});%
  \node[keytext] at (11.5mm,{-#1*6.5mm}) {#3};}
% A data store in the key: a small cylinder in the given rim and fill colours,
% brand/fillbrand for ETHOS.Data's stores, muted/fillsoft for external ones.
%   \keystore{6}{brand}{fillbrand}{Data store}
\newcommand{\keystore}[4]{%
  \begin{scope}[shift={(4.5mm,{-#1*6.5mm})}]
    \filldraw[draw=#2, fill=#3] (-3mm,1.5mm) -- (-3mm,-1.5mm)
      arc[start angle=180, end angle=360, x radius=3mm, y radius=0.8mm]
      -- (3mm,1.5mm)
      arc[start angle=0, end angle=180, x radius=3mm, y radius=0.8mm] -- cycle;
    \draw[draw=#2] (-3mm,1.5mm)
      arc[start angle=180, end angle=360, x radius=3mm, y radius=0.8mm];
  \end{scope}%
  \node[keytext] at (11.5mm,{-#1*6.5mm}) {#4};}
