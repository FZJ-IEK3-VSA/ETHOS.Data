% catalogue on the store; `catalog update-checkout` advances the checkout on
% the ICE-2 cluster computer to the latest release. Nothing synchronises by
% itself. Spelled -{}-push because a bare `--` is set as an en dash.

\node[artifact, text width=47mm, minimum height=17mm] (jugit) at (0,0) {JuGit\\internal source catalogue\\releases as tags};
\node[artifact, text width=47mm, minimum height=17mm] (github) at (8,0) {GitHub\\public catalogue\\releases as tags};
\node[actor, text width=47mm, minimum height=24mm] (cluster) at (0,-4) {ICE-2 cluster computer\\reader + maintainer tooling\\internal catalogue at a release\\shared caches};
\node[actor, text width=47mm, minimum height=24mm] (client) at (8,-4) {Workstation / CI runner\\package + reader\\pinned public release\\cache / repository bundles};
\node[store, text width=55mm, minimum height=22mm] (dcache) at (4,-9) {dCache\\authoritative published bytes,\\test datasets included,\\and the latest public catalogue};

\draw[flow] (jugit.south east) -- node[edgelabel,right]{update-checkout} (cluster.north east);
\draw[flow] (cluster.east) -- ++(21mm,0) |- node[edgelabel,pos=.25,right,align=left]{release -{}-push:\\public catalogue} (github.west);
