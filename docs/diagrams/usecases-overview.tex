% Role-based entry points; a role may be held by the same person.
\begin{tikzpicture}[node distance=12mm and 13mm]
\node[actor, text width=29mm] (user) {Data user};
\node[artifact, text width=86mm, right=of user, align=left] (get)
  {Discover collections and select task inputs\\
   Obtain data; configure local access\\
   Check integrity and diagnose unavailable files};
\node[actor, text width=29mm, below=22mm of user] (author) {Package\\maintainer};
\node[artifact, text width=86mm, right=of author, align=left] (integrate)
  {Wire collections into examples, workflows, and tests\\
   Stage development data and propose catalogue additions\\
   Keep verified repository copies; test deliberate local edits};
\node[actor, text width=29mm, below=22mm of author] (maint) {Catalogue\\maintainer};
\node[artifact, text width=86mm, right=of maint, align=left] (publish)
  {Review proposals and build the internal catalogue\\
   Upload and verify bytes; release public metadata\\
   Serve cluster snapshots; diagnose and repair tooling};
\draw[link] (user) -- (get);
\draw[link] (author) -- (integrate);
\draw[link] (maint) -- (publish);
\node[note, below=8mm of publish, text width=90mm]
  {Package maintainers propose changes. Catalogue maintainers accept and release them.\\
