% Dynamic view (arc42 6.3): from a proposal to a served release. One lane per
% place a step runs or writes: the package maintainer's checkout, the
% catalogue maintainer's own clone, GitHub, JuGit, dCache and the cluster
% computer. Time runs down; the numbers give the order. A step's arrows run
% sideways to what it writes, through lanes that are empty at that height.
% Coordinates are millimetres; the lanes split 0..163.
\begin{tikzpicture}[
  x=1mm, y=1mm,
  head/.style={anchor=north, minimum height=19mm, inner sep=1.2mm, font=\sffamily\footnotesize},
  lane/.style={draw=boundary, line width=0.5pt},
  % A step: a command, or (dashed) something a person does by hand.
  step/.style={draw=brand, fill=fillbrand, text=ink, rounded corners=1.2mm,
    align=left, font=\sffamily\scriptsize, inner sep=1.3mm, anchor=north west,
    text width=#1},
  step/.default=35.4mm,
  hand/.style={step=#1, dash pattern=on 3pt off 2pt},
  hand/.default=35.4mm,
  % Something a person posts or reviews on a host: an issue, a merge request.
  item/.style={draw=muted, fill=fillsoft, text=ink, rounded corners=1mm,
    align=center, font=\sffamily\scriptsize, inner sep=1.2mm, anchor=north west},
  % A cylinder #1 mm wide, placed by its centre; caps the same height at any width.
  st/.style={c4db, minimum height=12.5mm, inner sep=0.6mm,
    font=\sffamily\scriptsize, minimum width=#1mm, text width=\dimexpr#1mm-2mm\relax,

  lblbg/.style={c4rellabelbg},

  guard/.style={c4rellabelbg, text=muted, anchor=north west, inner sep=1pt},
]
